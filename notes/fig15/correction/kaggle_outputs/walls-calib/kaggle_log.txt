"""Fig. 15 correction, walls: the exact 3D field of a stored wall run, with no time grid and no barrier sampling.

Separation. V depends on x alone and the start packet is a product, so for all time
    psi(x, y, z, t) = chi(x, t) h(y, t) h(z, t),
h the free Gaussian of width sp:  h(y, t) = (2 pi sp^2)^(-1/4) (1 + i t / (2 sp^2))^(-1/2) exp(-y^2 / (4 sp^2 (1 + i t / (2 sp^2)))).

chi by superposition of stationary scattering states (hbar = m = 1, E = k^2 / 2, k the wave number on the left):
    chi(x, t) = Int dk / (2 pi) [ c+(k) psi+_k(x) + (k / kR) c-(k) psi-_k(x) ] e^{-i k^2 t / 2},
    psi+_k : e^{ikx} + r e^{-ikx} on the left,  t e^{i kR x} on the right            (incident from the left),
    psi-_k : t' e^{-ikx} on the left,  e^{-i kR x} + r' e^{i kR x} on the right      (incident from the right),
    kR = sqrt(k^2 + 2 bias),  c+- = <psi+-_k | chi0>  (the exact projections; normalisation <psi+_k|psi+_k'> = 2 pi delta(k - k'),
    <psi-_k|psi-_k'> = 2 pi (kR / k) delta(k - k')).
The two families with E > 0 are complete for a packet that starts on the left, up to right-incident states with -bias < E < 0,
which are evanescent on the left; their share is bounded by the start packet's share right of the first face (start_in_barrier).
Inside the structure psi is carried face to face by the exact flat-segment propagator; a bias ramp is cut into flat slices whose
convergence is measured (nslice).

Time enters only as e^{-iEt}: there is no time step and no grid in x, so neither the barrier sampling (D1) nor the side border
(D2) can arise. The k sum is a trapezoid on a uniform grid dk: it equals the exact integral periodised in x with period
2 pi / dk, so dk is chosen with 2 pi / dk far longer than the distance any part of the wave travels in the run.

Regions over the whole line: P_left(t) = P_left(0) - Int_0^t j(xl), P_right(t) = Int_0^t j(xr) (j the flux at the outer faces),
P_struct(t) from |chi|^2 on a fine grid between the faces; P_left + P_struct + P_right = 1 is checked, not imposed.
"""
import os, sys, json

def centres(sp):
    """Barrier centres as in rt3d2.centres (faces on odd multiples of 0.25)."""
    N, P, D = sp["N"], sp["P"], sp["D"]
    for c in (0.0, 0.25):
        cs = [c + (k - (N - 1) / 2) * P for k in range(N)]
        faces = [f for ck in cs for f in (ck - D / 2, ck + D / 2)]
        if all(abs((f - 0.25) / 0.5 - round((f - 0.25) / 0.5)) < 1e-9 for f in faces): return cs
    raise ValueError("no grid-aligned placement")


def x0_of(sp):
    cs = centres(sp); return cs[0] - sp["D"] / 2 - max(14.0, 4 * sp["sx"])


def segments(sp, nslice=2000):
    """Flat segments (a, b, V) tiling [xl, xr]. Unbiased walls are exact; a bias ramp is sliced, about nslice slices in all."""
    cs = centres(sp); D, V0, b = sp["D"], sp["V0"], sp.get("bias", 0.0) or 0.0
    xl, xr = cs[0] - D / 2, cs[-1] + D / 2
    faces = sorted({round(f, 12) for c in cs for f in (c - D / 2, c + D / 2)})
    segs = []
    for a, bb in zip(faces[:-1], faces[1:]):
        m = 0.5 * (a + bb); Vb = V0 if any(abs(m - c) < D / 2 for c in cs) else 0.0
        if not b: segs.append((a, bb, Vb)); continue
        n = max(4, int(np.ceil(nslice * (bb - a) / (xr - xl)))); xg = np.linspace(a, bb, n + 1)
        for u, v in zip(xg[:-1], xg[1:]): segs.append((u, v, Vb - b * (0.5 * (u + v) - xl) / (xr - xl)))
    return segs, xl, xr


def _prop(q, L):
    """cos(qL), sin(qL)/q, -q sin(qL) for complex q (q -> 0 limit handled)."""
    c = np.cos(q * L); small = np.abs(q) < 1e-9
    s = np.where(small, L, np.sin(q * L) / np.where(small, 1, q))
    return c, s, -q * q * s


class Scatter:
    """Both scattering families at the wave numbers k (left lead), and their values at given interior points."""

    def __init__(self, sp, k, xin, nslice=2000):
        self.sp = sp; self.k = k = np.asarray(k, float); bias = sp.get("bias", 0.0) or 0.0
        segs, xl, xr = segments(sp, nslice); self.xl, self.xr = xl, xr
        E = 0.5 * k * k; self.E = E; kR = np.sqrt(k * k + 2 * bias); self.kR = kR
        xin = np.asarray(xin, float); order = np.argsort(xin); xs = xin[order]
        # fundamental solutions u1: (psi, psi')(xl) = (1, 0), u2: (0, 1), carried right face by face
        p1, d1 = np.ones_like(k, complex), np.zeros_like(k, complex)
        p2, d2 = np.zeros_like(k, complex), np.ones_like(k, complex)
        U1 = np.zeros((len(xs), len(k)), complex); U2 = np.zeros_like(U1)
        j = 0
        for (a, b, V) in segs:
            q = np.sqrt(2 * (E - V) + 0j)
            while j < len(xs) and xs[j] < b:
                if xs[j] >= a:
                    c, s, _ = _prop(q, xs[j] - a); U1[j] = c * p1 + s * d1; U2[j] = c * p2 + s * d2
                j += 1
            c, s, m = _prop(q, b - a)
            p1, d1 = c * p1 + s * d1, m * p1 + c * d1
            p2, d2 = c * p2 + s * d2, m * p2 + c * d2
        inv = np.empty(len(xs), int); inv[order] = np.arange(len(xs))
        U1, U2 = U1[inv], U2[inv]
        # left-lead values of e^{ikx} (a) and e^{-ikx} (b) at xl
        ea, eb = np.exp(1j * k * xl), np.exp(-1j * k * xl)
        A = (ea, 1j * k * ea); B = (eb, -1j * k * eb)
        def at_r(v):            # (psi, psi')(xl) -> (psi, psi')(xr)
            return v[0] * p1 + v[1] * p2, v[0] * d1 + v[1] * d2
        def split(w):           # (psi, psi')(xr) -> amplitudes of e^{i kR x}, e^{-i kR x}
            ps, dp = w
            return 0.5 * (ps + dp / (1j * kR)) * np.exp(-1j * kR * xr), 0.5 * (ps - dp / (1j * kR)) * np.exp(1j * kR * xr)
        aR, bR = split(at_r(A)); aRb, bRb = split(at_r(B))
        r = -bR / bRb; t = aR + r * aRb                        # psi+: B_R = 0
        tp = 1 / bRb; rp = aRb / bRb                           # psi-: unit incoming e^{-i kR x}
        self.r, self.t, self.tp, self.rp = r, t, tp, rp
        # interior values of both families
        self.inP = (A[0] + r * B[0]) * U1 + (A[1] + r * B[1]) * U2
        self.inM = tp * (B[0] * U1 + B[1] * U2)
        self.T = (kR / k) * np.abs(t)**2; self.R = np.abs(r)**2

    def values(self, x, inside_idx=None):
        """psi+ and psi- on points x: analytic outside [xl, xr]; inside, from the precomputed interior table (inside_idx maps
        each interior point to its row)."""
        k, kR = self.k, self.kR; x = np.asarray(x, float)
        Pp = np.empty((len(x), len(k)), complex); Pm = np.empty_like(Pp)
        L = x < self.xl; Rr = x > self.xr; I = ~(L | Rr)
        if L.any():
            e = np.exp(1j * np.outer(x[L], k)); Pp[L] = e + self.r / e; Pm[L] = self.tp / e
        if Rr.any():
            e = np.exp(1j * np.outer(x[Rr], kR)); Pp[Rr] = self.t * e; Pm[Rr] = 1 / e + self.rp * e
        if I.any():
            Pp[I] = self.inP[inside_idx[I]]; Pm[I] = self.inM[inside_idx[I]]
        return Pp, Pm


def chi0(sp, x):
    s = sp["sx"]; k0 = sqrt(2 * sp["E"]); x0 = x0_of(sp)
    return (2 * pi * s * s) ** -0.25 * np.exp(-(x - x0)**2 / (4 * s * s) + 1j * k0 * x)


def G(sp, k):
    """Int e^{-ikx} chi0(x) dx over the whole line (closed form)."""
    s = sp["sx"]; k0 = sqrt(2 * sp["E"]); x0 = x0_of(sp)
    return (2 * pi * s * s) ** -0.25 * 2 * s * sqrt(pi) * np.exp(-s * s * (k - k0)**2 - 1j * (k - k0) * x0)


def kgrid(sp, dk=5e-4, nsig=8.5):
    k0 = sqrt(2 * sp["E"]); sk = 1 / (2 * sp["sx"])
    lo = max(dk, k0 - nsig * sk); hi = k0 + nsig * sk
    n = int(np.ceil((hi - lo) / dk)) + 1
    k = np.linspace(lo, hi, n); w = np.full(n, (hi - lo) / (n - 1)); w[0] *= 0.5; w[-1] *= 0.5
    return k, w


def simpson_w(n, h):
    if n % 2 == 0: raise ValueError("odd n")
    w = np.ones(n); w[1:-1:2] = 4; w[2:-1:2] = 2; return w * h / 3


class Exact:
    """The exact chi of one wall run. xs_out: the points where chi is wanted (the box's 0.5 a grid)."""

    def __init__(self, sp, xs_out, dk=5e-4, nslice=2000, hfine=0.01):
        self.sp = sp; k, w = kgrid(sp, dk); self.k, self.w = k, w
        segs, xl, xr = segments(sp, nslice)
        # fine grid over the structure (Simpson, per segment between true faces), for P_struct and the well
        cs = centres(sp); D = sp["D"]
        faces = sorted({round(f, 12) for c in cs for f in (c - D / 2, c + D / 2)})
        fine, fw, fwell = [], [], []
        for a, b in zip(faces[:-1], faces[1:]):
            n = int(np.ceil((b - a) / hfine)); n += (n % 2 == 1); n += 1
            xx = np.linspace(a, b, n); fine.append(xx); fw.append(simpson_w(n, (b - a) / (n - 1)))
            m = 0.5 * (a + b); fwell.append(np.full(n, not any(abs(m - c) < D / 2 for c in cs)))
        self.xf = np.concatenate(fine); self.wf = np.concatenate(fw); self.well = np.concatenate(fwell)
        # projection grid: chi0 right of xl (its tail), where psi differs from the left-lead form
        s = sp["sx"]; xt_hi = xl + 10 * s
        nt = int(np.ceil((xt_hi - xl) / hfine)); nt += (nt % 2 == 1); nt += 1
        self.xt = np.linspace(xl, xt_hi, nt); self.wt = simpson_w(nt, (xt_hi - xl) / (nt - 1))
        xs_out = np.asarray(xs_out, float); self.xs_out = xs_out
        inside = np.concatenate([self.xf, self.xt[(self.xt >= xl) & (self.xt <= xr)], xs_out[(xs_out >= xl) & (xs_out <= xr)]])
        S = Scatter(sp, k, inside, nslice); self.S = S
        nf = len(self.xf); nt_in = int(((self.xt >= xl) & (self.xt <= xr)).sum())
        idx_f = np.arange(nf)
        idx_t = np.full(len(self.xt), -1); idx_t[(self.xt >= xl) & (self.xt <= xr)] = nf + np.arange(nt_in)
        idx_o = np.full(len(xs_out), -1); io = (xs_out >= xl) & (xs_out <= xr); idx_o[io] = nf + nt_in + np.arange(io.sum())
        # projections c+ = <psi+|chi0>, c- = <psi-|chi0>: closed form for the left-lead form over the whole line, plus the
        # exact correction where chi0's tail meets the structure and beyond
        Gk, Gmk = G(sp, k), G(sp, -k)
        cP = Gk + np.conj(S.r) * Gmk
        cM = np.conj(S.tp) * Gmk
        Pp, Pm = S.values(self.xt, idx_t)
        e = np.exp(1j * np.outer(self.xt, k)); Lp = e + S.r / e; Lm = S.tp / e
        c0 = chi0(sp, self.xt) * self.wt
        cP = cP + ((np.conj(Pp) - np.conj(Lp)) * c0[:, None]).sum(0)
        cM = cM + ((np.conj(Pm) - np.conj(Lm)) * c0[:, None]).sum(0)
        self.cP, self.cM = cP, cM
        self.aP = w / (2 * pi) * cP; self.aM = w / (2 * pi) * (k / S.kR) * cM
        # completeness: the share of chi0 the two families carry (1 up to the evanescent right-incident states and k-range cut)
        self.carried = float((w / (2 * pi) * (np.abs(cP)**2 + (k / S.kR) * np.abs(cM)**2)).sum())
        # the share that ends on the right: outgoing right amplitude per k is c+ t + (k/kR) c- r'
        out = cP * S.t + (k / S.kR) * cM * S.rp
        self.T_inf = float((w / (2 * pi) * np.abs(out)**2 * (S.kR / k)).sum())
        self.T_packet_weights = float((w * np.abs(Gk)**2 * S.T).sum() / (w * np.abs(Gk)**2).sum())
        self.PL0 = 1 - 0.5 * erfc((xl - x0_of(sp)) / (sqrt(2) * s))   # Int_{-inf}^{xl} |chi0|^2
        # tables of psi on the output points and the fine structure grid, and at the two outer faces (value and slope)
        self.Bo = self._basis(xs_out, idx_o); self.Bf = self._basis(self.xf, idx_f)
        kk, kR = k, S.kR; el, er = np.exp(1j * kk * xl), np.exp(1j * kR * xr)
        self.fl = (self.aP * (el + S.r / el) + self.aM * (S.tp / el), self.aP * 1j * kk * (el - S.r / el) + self.aM * (-1j * kk) * S.tp / el)
        self.fr = (self.aP * S.t * er + self.aM * (1 / er + S.rp * er), self.aP * 1j * kR * S.t * er + self.aM * 1j * kR * (-1 / er + S.rp * er))

    def _basis(self, x, idx):
        Pp, Pm = self.S.values(x, idx)
        return Pp * self.aP + Pm * self.aM            # sum over k of this row times e^{-iEt} gives chi(x, t)

    def chi(self, times, which="out"):
        B = self.Bo if which == "out" else self.Bf
        ph = np.exp(-1j * np.outer(self.S.E, times))  # (nk, nt)
        return B @ ph                                  # (nx, nt)

    def _flux_cum(self, times, pairs, dt=0.02, chunk=1000):
        """Cumulative Int_0^t j dt' at the given times for several points at once; pairs = [(psi coefficients, psi' coefficients)].
        e^{-iEt} on the fine time grid is built by recurrence chunk to chunk (one exp table per run)."""
        tmax = float(np.max(times)); n = int(np.ceil(tmax / dt)); n += (n % 2 == 1); n = max(n, 2)
        tt = np.linspace(0, tmax, n + 1); h = tmax / n
        F = np.stack([c for pr in pairs for c in pr], 1)                   # (nk, 2 * npoints)
        J = np.empty((len(tt), len(pairs)))
        base = np.exp(-1j * np.outer(np.arange(chunk) * h, self.S.E))     # (chunk, nk)
        step = np.exp(-1j * self.S.E * chunk * h)
        cur = np.ones_like(self.S.E, complex)
        for a in range(0, len(tt), chunk):
            m = min(chunk, len(tt) - a)
            V = (base[:m] * cur) @ F
            for q in range(len(pairs)): J[a:a + m, q] = np.imag(np.conj(V[:, 2 * q]) * V[:, 2 * q + 1])
            cur = cur * step
        out = []
        for q in range(len(pairs)):
            j = J[:, q]; c = np.zeros(len(tt)); c[2::2] = np.cumsum(h / 3 * (j[0:-2:2] + 4 * j[1:-1:2] + j[2::2]))
            c[1::2] = c[0:-1:2] + 0.5 * h * (j[0:-1:2] + j[1::2]); out.append(np.interp(times, tt, c))
        return out

    def flux_integrals(self, times, dt=0.02):
        cl, cr = self._flux_cum(times, [self.fl, self.fr], dt)
        return self.PL0 - cl, cr

    def lead_pair(self, x0):
        k, kR, S = self.k, self.S.kR, self.S
        if x0 < self.S.xl:
            e = np.exp(1j * k * x0)
            return (self.aP * (e + S.r / e) + self.aM * (S.tp / e), self.aP * 1j * k * (e - S.r / e) - self.aM * 1j * k * S.tp / e)
        e = np.exp(1j * kR * x0)
        return (self.aP * S.t * e + self.aM * (1 / e + S.rp * e), self.aP * 1j * kR * S.t * e + self.aM * 1j * kR * (-1 / e + S.rp * e))

    def box_tails(self, times, xlo, xhi, dt=0.02):
        cl, ch = self._flux_cum(times, [self.lead_pair(xlo), self.lead_pair(xhi)], dt)
        s = self.sp["sx"]; x0 = x0_of(self.sp)
        lo0 = 0.5 * erfc((x0 - xlo) / (sqrt(2) * s)); hi0 = 0.5 * erfc((xhi - x0) / (sqrt(2) * s))
        return lo0 - cl, hi0 + ch

    def regions_and_tails(self, times, xlo, xhi, dt=0.02):
        """regions() and box_tails() with one shared time table."""
        cf = self.chi(times, "fine"); rho = np.abs(cf)**2
        Pst = (self.wf[:, None] * rho).sum(0); Pw = (self.wf[self.well][:, None] * rho[self.well]).sum(0)
        cl, cr, clo, chi_ = self._flux_cum(times, [self.fl, self.fr, self.lead_pair(xlo), self.lead_pair(xhi)], dt)
        PL, PR = self.PL0 - cl, cr
        s = self.sp["sx"]; x0 = x0_of(self.sp)
        lo = 0.5 * erfc((x0 - xlo) / (sqrt(2) * s)) - clo; hi = 0.5 * erfc((xhi - x0) / (sqrt(2) * s)) + chi_
        return dict(left=PL, right=PR, well=Pw, struct=Pst, closure=PL + PR + Pst), lo, hi

    def regions(self, times):
        cf = self.chi(times, "fine"); rho = np.abs(cf)**2
        Pst = (self.wf[:, None] * rho).sum(0); Pw = (self.wf[self.well][:, None] * rho[self.well]).sum(0)
        PL, PR = self.flux_integrals(times)
        return dict(left=PL, right=PR, well=Pw, struct=Pst, closure=PL + PR + Pst)


def h_free(y, t, s):
    a = 1 + 1j * t / (2 * s * s)
    return (2 * pi * s * s) ** -0.25 * a ** -0.5 * np.exp(-np.asarray(y)[:, None] ** 2 / (4 * s * s * a))

"""Fig. 15 correction, walls: exact_wall.py vectorised in JAX for the GPU (complex128). The same mathematics, line for line
(read exact_wall.py's docstring); only the arrays live on the device:

  - every point value is one gather and one fused elementwise expression over (points x wave numbers);

def _prop(q, L):

    s = jnp.where(small, L, jnp.sin(q * L) / jnp.where(small, 1.0, q))

class ExactJ:
    def __init__(self, sp, xs_out, dk=5e-4, nslice=2000, hfine=0.01):
        self.sp = sp; k_np, w_np = ew.kgrid(sp, dk); bias = sp.get("bias", 0.0) or 0.0
        NB = 1024; npad = (-len(k_np)) % NB                    # zero-weight padding: the sums are unchanged
        if npad:
            k_np = np.r_[k_np, k_np[-1] + dk * np.arange(1, npad + 1)]; w_np = np.r_[w_np, np.zeros(npad)]
        segs, xl, xr = ew.segments(sp, nslice); self.xl, self.xr = xl, xr
        k = jnp.asarray(k_np); w = jnp.asarray(w_np); E = 0.5 * k * k; kR = jnp.sqrt(k * k + 2 * bias)
        self.k, self.w, self.E, self.kR = k, w, E, kR
        sa = np.array([s[0] for s in segs]); sb = np.array([s[1] for s in segs]); sV = np.array([s[2] for s in segs])
        # fundamental solutions at every left face (scan over segments)
        def step(c, seg):
            a, b, V = seg; q = jnp.sqrt(2 * (E - V) + 0j); cc, ss, mm = _prop(q, b - a)
            p1, d1, p2, d2 = c
            return (cc * p1 + ss * d1, mm * p1 + cc * d1, cc * p2 + ss * d2, mm * p2 + cc * d2), c
        one = jnp.ones_like(k, dtype=c128); zero = jnp.zeros_like(k, dtype=c128)
        (P1, D1, P2, D2), faces = jax.lax.scan(step, (one, zero, zero, one), (jnp.asarray(sa), jnp.asarray(sb), jnp.asarray(sV)))
        self.faces = faces; self.seg = (sa, sb, sV)
        ea, eb = jnp.exp(1j * k * xl), jnp.exp(-1j * k * xl)
        A = (ea, 1j * k * ea); B = (eb, -1j * k * eb)
        def split(v):
            ps = v[0] * P1 + v[1] * P2; dp = v[0] * D1 + v[1] * D2
            return 0.5 * (ps + dp / (1j * kR)) * jnp.exp(-1j * kR * xr), 0.5 * (ps - dp / (1j * kR)) * jnp.exp(1j * kR * xr)
        aR, bR = split(A); aRb, bRb = split(B)
        r = -bR / bRb; t = aR + r * aRb; tp = 1 / bRb; rp = aRb / bRb
        self.r, self.t, self.tp, self.rp = r, t, tp, rp
        self.alphaP = (A[0] + r * B[0], A[1] + r * B[1]); self.alphaM = (tp * B[0], tp * B[1])
        # fine structure grid (Simpson per segment between true faces)
        cs = ew.centres(sp); D = sp["D"]
        fa = sorted({round(f, 12) for c in cs for f in (c - D / 2, c + D / 2)})
        fine, fw, fwell = [], [], []
        for a, b in zip(fa[:-1], fa[1:]):
            n = int(np.ceil((b - a) / hfine)); n += (n % 2 == 1); n += 1
            xx = np.linspace(a, b, n); fine.append(xx); fw.append(ew.simpson_w(n, (b - a) / (n - 1)))
            m = 0.5 * (a + b); fwell.append(np.full(n, not any(abs(m - c) < D / 2 for c in cs)))
        self.xf = np.concatenate(fine); self.wf = jnp.asarray(np.concatenate(fw)); self.well = np.concatenate(fwell)
        s = sp["sx"]; xt_hi = max(xr + 1.0, ew.x0_of(sp) + 8.6 * s)      # |chi0|^2 < 1e-16 of its peak beyond 8.6 sigma
        ht = 0.02; nt = int(np.ceil((xt_hi - xl) / ht)); nt = -(-nt // 512) * 512 + 1; xt_hi = xl + ht * (nt - 1)
        xt = np.linspace(xl, xt_hi, nt); wt = ew.simpson_w(nt, (xt_hi - xl) / (nt - 1))
        # projections: closed form + exact correction where chi0's tail meets the structure and beyond
        Gk = jnp.asarray(ew.G(sp, k_np)); Gmk = jnp.asarray(ew.G(sp, -k_np))
        Pp, Pm = self.values(xt)
        e = jnp.exp(1j * jnp.outer(jnp.asarray(xt), k)); Lp = e + r / e; Lm = tp / e
        c0 = jnp.asarray(ew.chi0(sp, xt) * wt)
        cP = Gk + jnp.conj(r) * Gmk + ((jnp.conj(Pp) - jnp.conj(Lp)) * c0[:, None]).sum(0)
        cM = jnp.conj(tp) * Gmk + ((jnp.conj(Pm) - jnp.conj(Lm)) * c0[:, None]).sum(0)
        del Pp, Pm, e, Lp, Lm
        self.aP = w / (2 * pi) * cP; self.aM = w / (2 * pi) * (k / kR) * cM
        self.carried = float((w / (2 * pi) * (jnp.abs(cP)**2 + (k / kR) * jnp.abs(cM)**2)).sum())
        out = cP * t + (k / kR) * cM * rp
        self.T_inf = float((w / (2 * pi) * jnp.abs(out)**2 * (kR / k)).sum())
        T = (kR / k) * jnp.abs(t)**2; g2 = w * jnp.abs(Gk)**2
        self.T_packet_weights = float((g2 * T).sum() / g2.sum())
        self.PL0 = 1 - 0.5 * erfc((xl - ew.x0_of(sp)) / (sqrt(2) * s))
        self.Bo = self.basis(np.asarray(xs_out, float)); self.Bf = self.basis(self.xf)
        el, er = jnp.exp(1j * k * xl), jnp.exp(1j * kR * xr)
        self.fl = (self.aP * (el + r / el) + self.aM * (tp / el), self.aP * 1j * k * (el - r / el) - self.aM * 1j * k * tp / el)
        self.fr = (self.aP * t * er + self.aM * (1 / er + rp * er), self.aP * 1j * kR * t * er + self.aM * 1j * kR * (-1 / er + rp * er))

    def values(self, x):
        """psi+ and psi- on the sorted points x (numpy), as (len(x), nk) device arrays: closed forms left and right of the
        structure, the face-to-face propagator only inside it."""
        k, kR = self.k, self.kR; x = np.asarray(x, float)
        assert np.all(np.diff(x) >= 0), "points must be sorted"
        nL = int((x < self.xl).sum()); nR = int((x > self.xr).sum()); xi = x[nL:len(x) - nR]
        parts_p, parts_m = [], []
        if nL:
            eL = jnp.exp(1j * jnp.asarray(x[:nL])[:, None] * k); parts_p.append(eL + self.r / eL); parts_m.append(self.tp / eL)
        if len(xi):
            sa, sb, sV = self.seg
            si = np.clip(np.searchsorted(sa, xi, side="right") - 1, 0, len(sa) - 1); idx = jnp.asarray(si); F = self.faces
            q = jnp.sqrt(2 * (self.E[None, :] - jnp.asarray(sV[si])[:, None]) + 0j)
            c, s_, _ = _prop(q, jnp.asarray(xi - sa[si])[:, None])
            U1 = c * F[0][idx] + s_ * F[1][idx]; U2 = c * F[2][idx] + s_ * F[3][idx]
            parts_p.append(self.alphaP[0] * U1 + self.alphaP[1] * U2); parts_m.append(self.alphaM[0] * U1 + self.alphaM[1] * U2)
        if nR:
            eR = jnp.exp(1j * jnp.asarray(x[len(x) - nR:])[:, None] * kR); parts_p.append(self.t * eR); parts_m.append(1 / eR + self.rp * eR)
        return jnp.concatenate(parts_p, 0), jnp.concatenate(parts_m, 0)

    def basis(self, x):
        Pp, Pm = self.values(x); return Pp * self.aP + Pm * self.aM

    def chi(self, times, which="out"):
        B = self.Bo if which == "out" else self.Bf
        return B @ jnp.exp(-1j * jnp.outer(self.E, jnp.asarray(times)))

    def lead_pair(self, x0):
        k, kR = self.k, self.kR
        if x0 < self.xl:
            e = jnp.exp(1j * k * x0)
            return (self.aP * (e + self.r / e) + self.aM * (self.tp / e), self.aP * 1j * k * (e - self.r / e) - self.aM * 1j * k * self.tp / e)
        e = jnp.exp(1j * kR * x0)
        return (self.aP * self.t * e + self.aM * (1 / e + self.rp * e), self.aP * 1j * kR * self.t * e + self.aM * 1j * kR * (-1 / e + self.rp * e))

    def _flux_cum(self, times, pairs, dt=0.02, chunk=4000):
        tmax = float(np.max(times)); n = int(np.ceil(tmax / dt)); n += (n % 2 == 1); n = max(n, 2)
        tt = np.linspace(0, tmax, n + 1); h = tmax / n
        F = jnp.stack([c for pr in pairs for c in pr], 1)
        base = jnp.exp(-1j * jnp.outer(jnp.arange(chunk) * h, self.E)); step = jnp.exp(-1j * self.E * chunk * h)
        cur = jnp.ones_like(self.E, dtype=c128); Js = []
        for a in range(0, len(tt), chunk):
            m = min(chunk, len(tt) - a); V = (base * cur) @ F
            Js.append(jnp.stack([jnp.imag(jnp.conj(V[:, 2 * q]) * V[:, 2 * q + 1]) for q in range(len(pairs))], 1)[:m])
            cur = cur * step
        J = np.asarray(jnp.concatenate(Js, 0)); out = []
        for q in range(len(pairs)):
            j = J[:, q]; c = np.zeros(len(tt)); c[2::2] = np.cumsum(h / 3 * (j[0:-2:2] + 4 * j[1:-1:2] + j[2::2]))
            c[1::2] = c[0:-1:2] + 0.5 * h * (j[0:-1:2] + j[1::2]); out.append(np.interp(times, tt, c))
        return out

    def regions_and_tails(self, times, xlo, xhi, dt=0.02):
        rho = jnp.abs(self.chi(times, "fine"))**2
        Pst = np.asarray((self.wf[:, None] * rho).sum(0))
        wm = jnp.asarray(self.well)[:, None]
        Pw = np.asarray((self.wf[:, None] * rho * wm).sum(0))
        cl, cr, clo, chi_ = self._flux_cum(times, [self.fl, self.fr, self.lead_pair(xlo), self.lead_pair(xhi)], dt)
        PL, PR = self.PL0 - cl, cr
        s = self.sp["sx"]; x0 = ew.x0_of(self.sp)
        lo = 0.5 * erfc((x0 - xlo) / (sqrt(2) * s)) - clo; hi = 0.5 * erfc((xhi - x0) / (sqrt(2) * s)) + chi_
        return dict(left=PL, right=PR, well=Pw, struct=Pst, closure=PL + PR + Pst), lo, hi

==================== correction/rebuild_wall.py ====================
"""Fig. 15 correction, walls: write a wall run's RT3C file from a given chi(x, t), in exactly the stored file's layout and frames.

Two sources of chi:
  'exact' : exact_wall.Exact (stationary-state superposition; the true 3D field psi = chi(x,t) h(y,t) h(z,t));
  'twin'  : the solver's own grid scheme in 1D (rt3d2.run1d: Strang split-step, 0.5 a, dt 0.01, sin^(1/8) border) for chi and the
            same scheme with V = 0 for h on the 128-point cross axis. The 3D scheme of rt3d2 separates exactly (V(x) alone, a product
            mask, a separable kinetic factor), so the twin must reproduce the stored file to its 8-bit steps: the writer check.
What the file holds, as rt3d2.run writes it: |psi| and arg psi on z = 0, y >= 0 over the box interior, 8-bit levels of each frame's
own peak; the x-profile (integral over y, z); region curves; for walls the 1D profile.
Usage: python3 rebuild_wall.py <stored file.bin> [exact|twin] [out.bin]
"""
import os, sys, json, struct, zlib, hashlib
import numpy as np
HERE = os.path.dirname(os.path.abspath(__file__)); sys.path.insert(0, HERE); sys.path.insert(0, os.path.dirname(HERE))
import read_rt3d, exact_wall as ew


def quantize(sl, ph):
    """The solver's own rule (rt3d2.run): levels of the frame's peak; phase levels; phase zeroed where the amplitude is < 3 levels."""
    sl = np.nan_to_num(sl); pk = float(sl.max()) or 1.0
    qa = np.clip(np.round(255 * sl / pk), 0, 255).astype(np.uint8)
    qp = (np.round((np.nan_to_num(ph) + np.pi) / (2 * np.pi) * 255) % 256).astype(np.uint8); qp[qa < 3] = 0
    return qa, qp, pk


def axis(n, dx=0.5):
    x = (np.arange(n) - n // 2) * dx; k = 2 * np.pi * np.fft.fftfreq(n, dx)
    W = max(4, n // 10); m = np.ones(n); e = np.arange(W); m[:W] = np.sin(0.5 * np.pi * e / W) ** 0.125; m[-W:] = m[:W][::-1]
    return x, k, m


def twin_fields(h):
    """chi_d(x, t_f) on the box grid and h_d(y, t_f) on the cross grid, by the stored run's own discrete scheme (complex128)."""
    sp = h["spec"]; dt = h["dt"]; per = int(round(h["frame_dt"] / dt)); n = h["n"]
    x, kx, mx = axis(sp["Nx"]); y, ky, my = axis(128)
    cs = h["centres"]; xl, xr = h["extent"]
    V = np.zeros_like(x)
    for c in cs: V += np.where(np.abs(x - c) < sp["D"] / 2, sp["V0"], 0.0)
    if sp.get("bias"): V += -sp["bias"] * np.clip((x - xl) / (xr - xl), 0, 1)
    k0 = np.sqrt(2 * sp["E"]); x0 = h["x0"]
    chi = np.exp(-(x - x0)**2 / (4 * sp["sx"]**2) + 1j * k0 * x); chi /= np.sqrt((abs(chi)**2).sum() * 0.5)
    g = np.exp(-y**2 / (4 * sp["sp"]**2)).astype(complex); g /= np.sqrt((abs(g)**2).sum() * 0.5)
    hv = np.exp(-0.5j * V * dt); kin = np.exp(-0.5j * kx**2 * dt); kiny = np.exp(-0.5j * ky**2 * dt)
    C = np.empty((n, len(x)), complex); H = np.empty((n, len(y)), complex)
    for f in range(n):
        if f:
            for _ in range(per):
                chi = hv * np.fft.ifft(kin * np.fft.fft(hv * chi)) * mx
                g = np.fft.ifft(kiny * np.fft.fft(g)) * my
        C[f] = chi; H[f] = g
    return x, C, y, H


def from_twin(h):
    x, C, y, H = twin_fields(h)
    sp = h["spec"]; i0 = int(round((h["x_first"] - x[0]) / 0.5)); nxs = h["nxs"]; j0 = 64
    S = (np.abs(H)**2).sum(1) * 0.5                       # transverse survival per axis
    slab = C[:, i0:i0 + nxs, None] * H[:, None, j0:] * H[:, j0, None, None]
    prof = np.abs(C)**2 * (S**2)[:, None]
    xl, xr = h["extent"]; cs = h["centres"]; D = sp["D"]
    inb = np.zeros_like(x, bool)
    for c in cs: inb |= np.abs(x - c) < D / 2
    reg = dict(left=x < xl, right=x > xr)
    if len(cs) > 1: reg["well"] = (x > cs[0] + D / 2) & (x < cs[-1] - D / 2) & ~inb
    t = np.arange(h["n"]) * h["frame_dt"]
    curves = [t, (np.abs(C)**2).sum(1) * 0.5 * S**2] + [(np.abs(C[:, m])**2).sum(1) * 0.5 * S**2 for _, m in sorted(reg.items())]
    return slab, prof, np.abs(C)**2, np.array(curves).T, ["t", "norm"] + sorted(reg)


def from_exact(h, dk=5e-4, nslice=2000):
    sp = dict(h["spec"]); xs = h["prof_x_first"] + np.arange(h["prof_nx"]) * h["step"]
    X = ew.Exact(sp, xs, dk=dk, nslice=nslice)
    t = np.arange(h["n"]) * h["frame_dt"]
    chi = X.chi(t).T                                         # (n, nx)
    i0 = int(round((h["x_first"] - xs[0]) / h["step"])); nxs = h["nxs"]
    yy = np.arange(64) * h["step"]; Hy = ew.h_free(yy, t, sp["sp"]).T      # (n, 64); h(0, t) = Hy[:, 0]
    slab = chi[:, i0:i0 + nxs, None] * Hy[:, None, :] * Hy[:, 0, None, None]
    prof = np.abs(chi)**2                                    # the transverse factor has norm 1 on the whole plane
    half = 0.5 * h["prof_nx"] * h["step"]
    R, lo, hi = X.regions_and_tails(t, -half, half)
    # the same regions as the stored file, counted over the box (|x| <= box half-length) so that "total in box" keeps its meaning;
    # the whole-line values are kept beside them
    curves = [t, 1 - lo - hi, R["left"] - lo, R["right"] - hi] + ([R["well"]] if h["spec"]["N"] > 1 else [])
    names = ["t", "norm", "left", "right"] + (["well"] if h["spec"]["N"] > 1 else [])
    extra = dict(left_line=R["left"], right_line=R["right"], struct=R["struct"], closure=R["closure"])
    return slab, prof, prof, np.array(curves).T, names, X, extra


def write(h, slab, prof, prof1d, curves, names, extra_head):
    A, P, peaks = [], [], []
    for f in range(len(slab)):
        qa, qp, pk = quantize(np.abs(slab[f]), np.angle(slab[f])); A.append(qa); P.append(qp); peaks.append(pk)
    head = dict(h)
    head.update(peaks=[round(p / peaks[0], 6) for p in peaks], peak0=peaks[0], curve_names=names,
                curves=np.round(curves, 7).tolist())
    head.update(extra_head)
    body = np.array(A).tobytes() + np.array(P).tobytes() + np.asarray(prof, np.float32).tobytes()
    if prof1d is not None: body += np.asarray(prof1d, np.float32).tobytes()
    hb = json.dumps(head, separators=(",", ":")).encode()
    return b"RT3C" + struct.pack("<I", len(hb)) + hb + zlib.compress(body, 6), np.array(A), np.array(P), peaks


def level_diff(a, b, phase=False):
    d = np.abs(a.astype(int) - b.astype(int))
    if phase: d = np.minimum(d, 256 - d)
    return d


if __name__ == "__main__":
    src = sys.argv[1]; mode = sys.argv[2] if len(sys.argv) > 2 else "exact"

def one(j, dev):

    t0 = time.time(); h = header(j); sp = h["spec"]; t = times_of(h)

    return assemble(j, chi, R, lo, hi, X.T_inf, X.T_packet_weights, X.carried, dev, time.time() - t0)

class Group:
    """Geometry shared by every wall of one group, and the jitted, vmapped core."""

    def __init__(self, jobs, dk=5e-4, nslice=2000, hfine=0.01):
        self.jobs = jobs; sp0 = jobs[0]["spec"]; h0 = header(jobs[0])
        segs, xl, xr = ew.segments(sp0, nslice); self.xl, self.xr = xl, xr
        self.sa = np.array([s[0] for s in segs]); self.sb = np.array([s[1] for s in segs])
        for j in jobs:                                       # the shared geometry is asserted, not assumed
            sg, a, b = ew.segments(j["spec"], nslice)
            assert len(sg) == len(segs) and a == xl and b == xr and np.array_equal(np.array([s[0] for s in sg]), self.sa) \
                and np.array_equal(np.array([s[1] for s in sg]), self.sb), "geometry differs inside the group"
            assert ew.x0_of(j["spec"]) == ew.x0_of(sp0) and j["spec"]["Nx"] == sp0["Nx"] and j["n"] == jobs[0]["n"]
        NB = 1024; self.K = max(-(-len(ew.kgrid(j["spec"], dk)[0]) // NB) * NB for j in jobs); self.dk = dk; self.nslice = nslice
        cs = ew.centres(sp0); D = sp0["D"]
        fa = sorted({round(f, 12) for c in cs for f in (c - D / 2, c + D / 2)})
        fine, fw, fwell = [], [], []
        for a, b in zip(fa[:-1], fa[1:]):
            n = int(np.ceil((b - a) / hfine)); n += (n % 2 == 1); n += 1
            xx = np.linspace(a, b, n); fine.append(xx); fw.append(ew.simpson_w(n, (b - a) / (n - 1)))
            m = 0.5 * (a + b); fwell.append(np.full(n, not any(abs(m - c) < D / 2 for c in cs)))
        self.xf = np.concatenate(fine); wf = np.concatenate(fw); well = np.concatenate(fwell)
        s = sp0["sx"]; xt_hi = max(xr + 1.0, ew.x0_of(sp0) + 8.6 * s)
        ht = 0.02; nt = int(np.ceil((xt_hi - xl) / ht)); nt = -(-nt // 512) * 512 + 1; xt_hi = xl + ht * (nt - 1)
        self.xt = np.linspace(xl, xt_hi, nt); self.wt = ew.simpson_w(nt, (xt_hi - xl) / (nt - 1))
        self.xo = xs_of(h0); self.half = 0.5 * h0["prof_nx"] * 0.5
        self.PL0 = 1 - 0.5 * erfc((xl - ew.x0_of(sp0)) / (sqrt(2) * s))
        sa, sb = self.sa, self.sb
        def pts(x):
            x = np.asarray(x, float); nL = int((x < xl).sum()); nR = int((x > xr).sum()); xi = x[nL:len(x) - nR]
            si = np.clip(np.searchsorted(sa, xi, side="right") - 1, 0, len(sa) - 1)
            return x[:nL], xi, si, xi - sa[si], x[len(x) - nR:]
        Pt, Pf, Po = pts(self.xt), pts(self.xf), pts(self.xo)
        xt_j, wf_j, wfw_j = jnp.asarray(self.xt), jnp.asarray(wf), jnp.asarray(wf * well)
        def lead(x0, k, kR, aP, aM, r, t, tp, rp):
            if x0 < xl:
                e = jnp.exp(1j * k * x0)
                return aP * (e + r / e) + aM * (tp / e), aP * 1j * k * (e - r / e) - aM * 1j * k * tp / e
            e = jnp.exp(1j * kR * x0)
            return aP * t * e + aM * (1 / e + rp * e), aP * 1j * kR * t * e + aM * 1j * kR * (-1 / e + rp * e)

        def core(k, w, bias, V, Gk, Gmk, c0, times):
            E = 0.5 * k * k; kR = jnp.sqrt(k * k + 2 * bias)
            def step(c, seg):
                a, b, v = seg; q = jnp.sqrt(2 * (E - v) + 0j); cc, ss, mm = ej._prop(q, b - a)
                p1, d1, p2, d2 = c
                return (cc * p1 + ss * d1, mm * p1 + cc * d1, cc * p2 + ss * d2, mm * p2 + cc * d2), c
            one_ = jnp.ones_like(k, dtype=c128); zero = jnp.zeros_like(k, dtype=c128)
            (P1, D1, P2, D2), F = jax.lax.scan(step, (one_, zero, zero, one_), (jnp.asarray(sa), jnp.asarray(sb), V))
            ea, eb = jnp.exp(1j * k * xl), jnp.exp(-1j * k * xl)
            A = (ea, 1j * k * ea); B = (eb, -1j * k * eb)
            def split(v):
                ps = v[0] * P1 + v[1] * P2; dp = v[0] * D1 + v[1] * D2
                return 0.5 * (ps + dp / (1j * kR)) * jnp.exp(-1j * kR * xr), 0.5 * (ps - dp / (1j * kR)) * jnp.exp(1j * kR * xr)
            aR, bR = split(A); aRb, bRb = split(B)
            r = -bR / bRb; t = aR + r * aRb; tp = 1 / bRb; rp = aRb / bRb
            alP = (A[0] + r * B[0], A[1] + r * B[1]); alM = (tp * B[0], tp * B[1])
            def values(p):
                xL, xi, si, dx, xR = p; pp, pm = [], []
                if len(xL):
                    eL = jnp.exp(1j * jnp.asarray(xL)[:, None] * k); pp.append(eL + r / eL); pm.append(tp / eL)
                if len(xi):
                    idx = jnp.asarray(si); q = jnp.sqrt(2 * (E[None, :] - V[idx][:, None]) + 0j)
                    c, s_, _ = ej._prop(q, jnp.asarray(dx)[:, None])
                    U1 = c * F[0][idx] + s_ * F[1][idx]; U2 = c * F[2][idx] + s_ * F[3][idx]
                    pp.append(alP[0] * U1 + alP[1] * U2); pm.append(alM[0] * U1 + alM[1] * U2)
                if len(xR):
                    eR = jnp.exp(1j * jnp.asarray(xR)[:, None] * kR); pp.append(t * eR); pm.append(1 / eR + rp * eR)
                return jnp.concatenate(pp, 0), jnp.concatenate(pm, 0)
            Pp, Pm = values(Pt)
            e = jnp.exp(1j * jnp.outer(xt_j, k)); Lp = e + r / e; Lm = tp / e
            cP = Gk + jnp.conj(r) * Gmk + ((jnp.conj(Pp) - jnp.conj(Lp)) * c0[:, None]).sum(0)
            cM = jnp.conj(tp) * Gmk + ((jnp.conj(Pm) - jnp.conj(Lm)) * c0[:, None]).sum(0)
            aP = w / (2 * pi) * cP; aM = w / (2 * pi) * (k / kR) * cM
            carried = (w / (2 * pi) * (jnp.abs(cP)**2 + (k / kR) * jnp.abs(cM)**2)).sum()
            out = cP * t + (k / kR) * cM * rp
            T_inf = (w / (2 * pi) * jnp.abs(out)**2 * (kR / k)).sum()
            g2 = w * jnp.abs(Gk)**2; T_w = (g2 * (kR / k) * jnp.abs(t)**2).sum() / g2.sum()
            ph = jnp.exp(-1j * jnp.outer(E, times))
            def basis(p):
                a, b = values(p); return a * aP + b * aM
            chi_o = basis(Po) @ ph
            rho = jnp.abs(basis(Pf) @ ph)**2
            Pst = (wf_j[:, None] * rho).sum(0); Pw = (wfw_j[:, None] * rho).sum(0)
            el, er = jnp.exp(1j * k * xl), jnp.exp(1j * kR * xr)
            fl = (aP * (el + r / el) + aM * (tp / el), aP * 1j * k * (el - r / el) - aM * 1j * k * tp / el)
            fr = (aP * t * er + aM * (1 / er + rp * er), aP * 1j * kR * t * er + aM * 1j * kR * (-1 / er + rp * er))
            lo = lead(-self.half, k, kR, aP, aM, r, t, tp, rp); hi = lead(self.half, k, kR, aP, aM, r, t, tp, rp)
            return dict(T_inf=T_inf, T_w=T_w, carried=carried, chi=chi_o, Pst=Pst, Pw=Pw, E=E,
                        fl0=fl[0], fl1=fl[1], fr0=fr[0], fr1=fr[1], lo0=lo[0], lo1=lo[1], hi0=hi[0], hi1=hi[1])
        self.fn = jax.jit(jax.vmap(core))
        S = len(sa); nt_, nf, no = len(self.xt), len(self.xf), len(self.xo)
        self.bytes_per_wall = 16 * self.K * (8 * nt_ + 4 * nf + 4 * no + 3 * 200 + 5 * S + 40)
        budget = float(globals().get("MEM_BUDGET", 6e9))
        self.B = int(max(1, min(int(globals().get("BATCH_MAX", 8)), budget // self.bytes_per_wall)))

    def inputs(self, j):
        sp = j["spec"]; k, w = ew.kgrid(sp, self.dk); npad = self.K - len(k)
        if npad: k = np.r_[k, k[-1] + self.dk * np.arange(1, npad + 1)]; w = np.r_[w, np.zeros(npad)]
        V = np.array([s[2] for s in ew.segments(sp, self.nslice)[0]])
        return (k, w, float(sp.get("bias", 0.0) or 0.0), V, ew.G(sp, k), ew.G(sp, -k), ew.chi0(sp, self.xt) * self.wt,
                times_of(header(j)))

    def run(self, batch, dev):
        """batch: list of jobs (len <= B; padded to B by repeating the last, results dropped). Returns [(blob, rec, A)]."""
        t0 = time.time(); jobs = batch + [batch[-1]] * (self.B - len(batch))
        ins = [self.inputs(j) for j in jobs]
        args = [jax.device_put(np.stack([np.asarray(x[i]) for x in ins]), dev) for i in range(8)]
        O = self.fn(*args); O = {kk: v for kk, v in O.items()}
        out = []; tb = time.time() - t0
        for b, j in enumerate(batch):
            t1 = time.time(); t = times_of(header(j)); sp = j["spec"]
            ns = SimpleNamespace(E=O["E"][b])
            pairs = [(O["fl0"][b], O["fl1"][b]), (O["fr0"][b], O["fr1"][b]), (O["lo0"][b], O["lo1"][b]), (O["hi0"][b], O["hi1"][b])]
            cl, cr, clo, chi_ = ej.ExactJ._flux_cum(ns, t, pairs, 0.02)
            PL, PR = self.PL0 - cl, cr; Pst = np.asarray(O["Pst"][b])
            s = sp["sx"]; x0 = ew.x0_of(sp)
            lo = 0.5 * erfc((x0 + self.half) / (sqrt(2) * s)) - clo; hi = 0.5 * erfc((self.half - x0) / (sqrt(2) * s)) + chi_
            R = dict(left=PL, right=PR, well=np.asarray(O["Pw"][b]), struct=Pst, closure=PL + PR + Pst)
            out.append(assemble(j, np.asarray(O["chi"][b]).T, R, lo, hi, float(O["T_inf"][b]), float(O["T_w"][b]),
                                float(O["carried"][b]), dev, tb / len(batch) + time.time() - t1))
        return out

"""Reads Fig. 15's 3D run files ('RT3C', written by rt3d_run.py on Kaggle) and prints the checks of each run.
Usage: python3 read_rt3d.py <file.bin | folder | rt3d.tar> ..."""
import json, struct, zlib, sys, os, tarfile, io
import numpy as np


def read(blob):
    assert blob[:4] == b"RT3C", "not an RT3C file"
    H = struct.unpack("<I", blob[4:8])[0]
    h = json.loads(blob[8:8 + H])
    raw = zlib.decompress(blob[8 + H:])
    n, nxs, nr, nx = h["n"], h["nxs"], h["nr"], h["prof_nx"]
    a = n * nxs * nr
    amp = np.frombuffer(raw, np.uint8, a, 0).reshape(n, nxs, nr)
    phase = np.frombuffer(raw, np.uint8, a, a).reshape(n, nxs, nr)
    prof = np.frombuffer(raw, np.float32, n * nx, 2 * a).reshape(n, nx)
    prof1d = np.frombuffer(raw, np.float32, n * nx, 2 * a + 4 * n * nx).reshape(n, nx) if h.get("has_1d") else None
    return h, amp, phase, prof, prof1d


def blobs(paths):
    for p in paths:
        if p.endswith(".tar"):
            with tarfile.open(p) as tf:
                for m in tf.getmembers():
                    if m.name.endswith(".bin"): yield m.name, tf.extractfile(m).read()
        elif os.path.isdir(p):
            for f in sorted(os.listdir(p)):
                if f.endswith(".bin"): yield f, open(os.path.join(p, f), "rb").read()
        else:
            yield os.path.basename(p), open(p, "rb").read()
